        // 3. ROBUST TEXT SPLITTER
        function wrapTextNodes(element) {
            const walker = document.createTreeWalker(element, NodeFilter.SHOW_TEXT, null, false);
            const nodesToReplace = [];
            while (walker.nextNode()) {
                if (walker.currentNode.nodeValue.trim() !== '') nodesToReplace.push(walker.currentNode);
            }

            nodesToReplace.forEach(node => {
                const text = node.nodeValue;
                const fragment = document.createDocumentFragment();
                const words = text.split(/(\s+)/); 
                
                words.forEach(word => {
                    if (word.trim() === '') {
                        fragment.appendChild(document.createTextNode(word));
                    } else {
                        const wordSpan = document.createElement('span');
                        wordSpan.style.display = 'inline-block';
                        wordSpan.style.whiteSpace = 'nowrap';
                        wordSpan.className = 'split-line';
                        
                        const charSpan = document.createElement('span');
                        charSpan.className = 'char';
                        charSpan.textContent = word; // Doing word-level split for heavy impact
                        wordSpan.appendChild(charSpan);
                        fragment.appendChild(wordSpan);
                    }
                });
                node.parentNode.replaceChild(fragment, node);
            });
        }
        document.querySelectorAll('.kinetic-text').forEach(el => wrapTextNodes(el));

        // 4. WEBGL GENERATIVE CONCRETE / GRIT SHADER
        // Replaces the neon orb with a high-performance 2D noise displacement simulating physical texture
        const initWebGL = () => {
            const container = document.getElementById('webgl-noise');
            const scene = new THREE.Scene();
            const camera = new THREE.OrthographicCamera(-1, 1, 1, -1, 0, 1);
            
            const renderer = new THREE.WebGLRenderer({ alpha: true, antialias: false, powerPreference: "high-performance" });

                    gl_Position = vec4(position, 1.0);

            `;

            const fragmentShader = `
                uniform float uTime;
                uniform float uScroll;
                varying vec2 vUv;

                // Hash function for grain
                float hash(vec2 p) {
                    p = fract(p * vec2(123.34, 456.21));
                    p += dot(p, p + 45.32);
                    return fract(p.x * p.y);
                }

                // Simplex noise (simplified 2D)
                vec3 permute(vec3 x) { return mod(((x*34.0)+1.0)*x, 289.0); }
                float snoise(vec2 v){
                    const vec4 C = vec4(0.211324865405187, 0.366025403784439, -0.577350269189626, 0.024390243902439);
                    vec2 i  = floor(v + dot(v, C.yy) );
                    vec2 x0 = v -   i + dot(i, C.xx);
                    vec2 i1;
                    i1 = (x0.x > x0.y) ? vec2(1.0, 0.0) : vec2(0.0, 1.0);
                    vec4 x12 = x0.xyxy + C.xxzz;
                    x12.xy -= i1;
                    i = mod(i, 289.0);
                    vec3 p = permute( permute( i.y + vec3(0.0, i1.y, 1.0 )) + i.x + vec3(0.0, i1.x, 1.0 ));
                    vec3 m = max(0.5 - vec3(dot(x0,x0), dot(x12.xy,x12.xy), dot(x12.zw,x12.zw)), 0.0);
                    m = m*m ;
                    m = m*m ;
                    vec3 x = 2.0 * fract(p * C.www) - 1.0;
                    vec3 h = abs(x) - 0.5;
                    vec3 ox = floor(x + 0.5);
                    vec3 a0 = x - ox;
                    m *= 1.79284291400159 - 0.85373472095314 * ( a0*a0 + h*h );
                    vec3 g;
                    g.x  = a0.x  * x0.x  + h.x  * x0.y;
                    g.yz = a0.yz * x12.xz + h.yz * x12.yw;
                    return 130.0 * dot(m, g);
                }

                void main() {

                    float n2 = snoise(uv * 6.0 - vec2(uScroll * 0.2, uTime * 0.02));
                    
                    // High frequency grain
                    float grain = hash(uv * 500.0 + uTime);
                    
                    float pattern = (n1 * 0.6 + n2 * 0.4);
                    pattern = smoothstep(-0.5, 0.5, pattern);
                    
                    // Construct brutalist concrete grit colors
                    vec3 darkConcrete = vec3(0.06, 0.06, 0.06); // #111111 approximation
                    vec3 lightConcrete = vec3(0.3, 0.3, 0.3);
                    
                    vec3 color = mix(darkConcrete, lightConcrete, pattern);
                    color -= grain * 0.15; // Apply grit

                    gl_FragColor = vec4(color, 1.0);
                }
            `;
